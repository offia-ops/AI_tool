Google Apps Script 程式碼 (點此檢視)

請將以下最新版程式碼貼入您的 Google Apps Script：

function setup() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  sheet.appendRow(["No.1", "2026-08-14", "系統測試", "100", "200", "100"]);
}

function doPost(e) {
  try {
    if (!e || !e.parameter) {
      return ContentService.createTextOutput("Error: 參數空白");
    }
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    
    var sn = "No." + sheet.getLastRow();
    
    // 日期格式：僅年月日 (例如 2026-08-14)
    var date = Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy-MM-dd");
    
    var name = e.parameter['飾品名稱'] || '';
    var cost = e.parameter['總成本'] || '';
    var price = e.parameter['建議售價'] || '';
    var profit = e.parameter['預估淨利'] || '';
    
    // 依序寫入: 序號(A欄), 日期(B欄), 名稱, 總成本, 建議售價, 預估淨利
    sheet.appendRow([sn, date, name, cost, price, profit]);
    
    return ContentService.createTextOutput("Success");
  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.toString());
  }
}
